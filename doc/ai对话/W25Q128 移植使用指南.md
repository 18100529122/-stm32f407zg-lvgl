### **2. 实现底层接口 (driver_w25qxx_interface.c)**
你需要根据 STM32F407ZG 的硬件连接（SPI1, PB14 为 CS）实现接口函数。以下是推荐的实现方式：

```c
#include "driver_w25qxx_interface.h"
#include "spi.h"
#include "main.h"
#include "bsp.h" // 包含 delay_us/delay_ms
#include <stdarg.h>

/**
 * @brief SPI 读写接口实现
 */
uint8_t w25qxx_interface_spi_qspi_write_read(uint8_t *instruction, uint8_t instruction_len, 
                                            uint8_t *data_out, uint32_t data_out_len, 
                                            uint8_t *data_in, uint32_t data_in_len)
{
    HAL_StatusTypeDef res;

    // 1. 拉低片选 (CS Low) - 根据引脚定义 PB14
    HAL_GPIO_WritePin(SPI1_CS_GPIO_Port, SPI1_CS_Pin, GPIO_PIN_RESET);

    // 2. 发送指令
    if (instruction_len > 0) {
        res = HAL_SPI_Transmit(&hspi1, instruction, instruction_len, 1000);
        if (res != HAL_OK) goto error;
    }

    // 3. 发送数据 (如果有)
    if (data_out_len > 0) {
        res = HAL_SPI_Transmit(&hspi1, data_out, data_out_len, 1000);
        if (res != HAL_OK) goto error;
    }

    // 4. 接收数据 (如果有)
    if (data_in_len > 0) {
        res = HAL_SPI_Receive(&hspi1, data_in, data_in_len, 1000);
        if (res != HAL_OK) goto error;
    }

    // 5. 拉高片选 (CS High)
    HAL_GPIO_WritePin(SPI1_CS_GPIO_Port, SPI1_CS_Pin, GPIO_PIN_SET);
    return 0;

error:
    HAL_GPIO_WritePin(SPI1_CS_GPIO_Port, SPI1_CS_Pin, GPIO_PIN_SET);
    return 1;
}

/**
 * @brief 延时与打印接口
 */
void w25qxx_interface_delay_ms(uint32_t ms) { delay_ms(ms); }
void w25qxx_interface_delay_us(uint32_t us) { delay_us(us); }
void w25qxx_interface_debug_print(const char *const fmt, ...) {
    va_list args;
    va_start(args, fmt);
    vprintf(fmt, args);
    va_end(args);
}

// SPI 初始化/反初始化可留空（如果已经在 main.c 中由 MX_SPI1_Init 完成）
uint8_t w25qxx_interface_spi_qspi_init(void) { return 0; }
uint8_t w25qxx_interface_spi_qspi_deinit(void) { return 0; }
```

### **3. 初始化与基本使用**
在你的应用代码中（如 `main.c` 或测试文件），按照以下流程初始化 W25Q128：

```c
#include "driver_w25qxx_basic.h"

void W25Q128_Test(void) {
    uint8_t res;
    uint8_t manufacturer, device_id;
    uint8_t test_data[16] = "Hello W25Q128!";
    uint8_t read_buf[16];

    // 1. 初始化 (W25Q128, 标准 SPI 接口)
    res = w25qxx_basic_init(W25Q128, W25QXX_INTERFACE_SPI, W25QXX_BOOL_FALSE);
    if (res != 0) {
        printf("W25Q128 Init Failed!\n");
        return;
    }

    // 2. 读取 ID 验证连接
    w25qxx_basic_get_id(&manufacturer, &device_id);
    printf("Manufacturer: 0x%02X, Device ID: 0x%02X\n", manufacturer, device_id);

    // 3. 读写测试 (注意：写之前通常需要擦除扇区，basic_write 内部已处理部分逻辑)
    // 写入地址 0x000000
    w25qxx_basic_write(0x000000, test_data, 16);
    
    // 读取验证
    w25qxx_basic_read(0x000000, read_buf, 16);
    printf("Read Data: %s\n", read_buf);
}
```
